cmake_minimum_required(VERSION 3.25)
project(BiglyBTC_Engine_Probe VERSION 0.1.0 LANGUAGES CXX)
include(../cmake/Toolchain.cmake)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
find_package(LibtorrentRasterbar 2.0.12 EXACT CONFIG REQUIRED)
add_executable(biglybtc-engine-probe main.cpp)
target_link_libraries(biglybtc-engine-probe PRIVATE LibtorrentRasterbar::torrent-rasterbar)
target_compile_options(biglybtc-engine-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(biglybtc-multifile-peer multifile_peer.cpp)
add_library(biglybtc-original-layout STATIC ../../../native/core/src/torrent_layout.cpp ../../../native/core/src/bencode.cpp)
target_include_directories(biglybtc-original-layout PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../../native/core/include")
target_compile_options(biglybtc-original-layout PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(biglybtc-multifile-peer PRIVATE biglybtc-original-layout LibtorrentRasterbar::torrent-rasterbar)
target_compile_options(biglybtc-multifile-peer PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
