cmake_minimum_required(VERSION 3.25)
project(BiglyBTC_Compatibility_Probe VERSION 0.1.0 LANGUAGES CXX)
include(../cmake/Toolchain.cmake)
find_package(OpenSSL 3.5.5 EXACT REQUIRED COMPONENTS Crypto)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../native/core" native-core)
add_library(biglybtc_formats STATIC crypto.cpp)
target_link_libraries(biglybtc_formats PUBLIC BiglyBTC::Core OpenSSL::Crypto)
target_compile_options(biglybtc_formats PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(biglybtc-format-probe main.cpp)
target_link_libraries(biglybtc-format-probe PRIVATE biglybtc_formats)
target_compile_options(biglybtc-format-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(biglybtc-format-tests tests.cpp)
target_link_libraries(biglybtc-format-tests PRIVATE biglybtc_formats)
target_compile_options(biglybtc-format-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded-formats COMMAND biglybtc-format-tests)
set_tests_properties(bounded-formats PROPERTIES TIMEOUT 30)
add_test(NAME java-format-fixtures COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/differential.py"
  --probe "$<TARGET_FILE:biglybtc-format-probe>")
set_tests_properties(java-format-fixtures PROPERTIES TIMEOUT 60)
