cmake_minimum_required(VERSION 3.28)
project(OwnedSocksCancellation LANGUAGES CXX)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IS_ABSOLUTE "${CANCEL_PROVIDER_INSTALL}" OR NOT IS_ABSOLUTE "${CANCEL_PROVIDER_SOURCE}")
  message(FATAL_ERROR "Explicit absolute CANCEL_PROVIDER_INSTALL and CANCEL_PROVIDER_SOURCE required")
endif()
foreach(header proxy_base.hpp socks5_stream.hpp)
  file(SHA256 "${CANCEL_PROVIDER_SOURCE}/include/libtorrent/${header}" source_hash)
  file(SHA256 "${CANCEL_PROVIDER_INSTALL}/include/libtorrent/${header}" installed_hash)
  if(NOT source_hash STREQUAL installed_hash)
    message(FATAL_ERROR "Provider source/installed header differs: ${header}")
  endif()
endforeach()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY FALSE)
find_package(LibtorrentRasterbar CONFIG REQUIRED PATHS "${CANCEL_PROVIDER_INSTALL}/lib/cmake/LibtorrentRasterbar" NO_DEFAULT_PATH)
find_package(Threads REQUIRED)
add_executable(owned-socks-cancellation cancel_fixture.cpp "${CANCEL_PROVIDER_SOURCE}/src/proxy_base.cpp")
# proxy_base's constructor/destructor are intentionally not public DSO exports;
# compile only their exact provider source with the selected installed SDK.
target_link_libraries(owned-socks-cancellation PRIVATE LibtorrentRasterbar::torrent-rasterbar Threads::Threads)
message(STATUS "CANCEL_PROVIDER_CONFIG=${LibtorrentRasterbar_CONFIG}")

enable_testing()
add_test(NAME socks-cancellation COMMAND owned-socks-cancellation --expect-fixed)
set_tests_properties(socks-cancellation PROPERTIES TIMEOUT 40)
