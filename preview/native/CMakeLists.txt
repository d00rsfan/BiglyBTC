cmake_minimum_required(VERSION 3.25)
project(BiglyBTC VERSION 0.1.0 LANGUAGES CXX)
include(cmake/Toolchain.cmake)
if(WIN32)
  add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(MSVC)
  add_compile_options(/utf-8 /Zc:__cplusplus)
endif()
option(BIGLYBTC_DESKTOP "Build the experimental Qt desktop using native services" OFF)
option(BIGLYBTC_LOCAL_ENGINE "Build the experimental owned-loopback engine runner" OFF)
option(BIGLYBTC_DEB "Enable the experimental Linux DEB packaging target" OFF)
option(BIGLYBTC_BUNDLE_QT "Bundle third-party desktop runtime libraries in the DEB" ON)
set(BIGLYBTC_RELEASE_SDK_SOURCE "" CACHE PATH "Source tree for the native release SDK")
if(BIGLYBTC_RELEASE_SDK_SOURCE)
  if(NOT BIGLYBTC_DESKTOP OR NOT BIGLYBTC_LOCAL_ENGINE OR NOT (BIGLYBTC_DEB OR APPLE))
    message(FATAL_ERROR "The release runtime requires the complete DEB or macOS desktop build")
  endif()
  set(BIGLYBTC_PERSISTENT_RUNTIME_MANIFEST
    "${CMAKE_CURRENT_SOURCE_DIR}/engine/runtime-binding/provider-original-rate-policy-runtime.json")
  set(BIGLYBTC_PERSISTENT_PUBLIC_CHAT_RUNTIME ON)
  set(BIGLYBTC_PERSISTENT_TAG_RATE_RUNTIME ON)
  set(BIGLYBTC_PERSISTENT_SHARED_QUOTA_RUNTIME ON)
  set(BIGLYBTC_SHARED_QUOTA_ALL_ROLES ON)
  set(BIGLYBTC_PERSISTENT_MIXED_I2P_RUNTIME ON)
  set(BIGLYBTC_PRIVATE_I2P_PROVIDER_OWNER ON)
  set(BIGLYBTC_MSGSYNC_MESSAGE_CORE ON)
  set(BIGLYBTC_ORIGINAL_MLDHT_IDENTITY_READER ON)
  set(BIGLYBTC_SHARED_UDP_SDK_SOURCE "${BIGLYBTC_RELEASE_SDK_SOURCE}")
endif()
set(BIGLYBTC_QT_REQUIREMENT 6.12.0 EXACT)
if(BIGLYBTC_DEB AND NOT BIGLYBTC_BUNDLE_QT)
  # Ubuntu 26.04 supplies Qt 6.10.2 through APT.
  set(BIGLYBTC_QT_REQUIREMENT 6.10.2)
endif()
include(CTest)
if(MINGW AND BIGLYBTC_DESKTOP)
  # Qt's MinGW entry-point target must first be imported in the common scope.
  find_package(Qt6 ${BIGLYBTC_QT_REQUIREMENT} REQUIRED COMPONENTS Core Network Widgets GLOBAL)
endif()
if(BIGLYBTC_DESKTOP)
  set(BIGLYBTC_SDK_TESTS OFF CACHE BOOL "SDK tested separately")
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../plugin-sdk" "plugin-sdk" EXCLUDE_FROM_ALL)
endif()
add_subdirectory(core)
add_subdirectory(platform)
add_subdirectory(adapters)
add_subdirectory(plugins/BiglyBTC-plugin-azrating)
if(BIGLYBTC_DESKTOP AND NOT BIGLYBTC_LOCAL_ENGINE)
  message(FATAL_ERROR "The experimental desktop currently requires the local engine adapter")
endif()
if(BIGLYBTC_LOCAL_ENGINE)
  add_subdirectory(engine)
  if(BIGLYBTC_DESKTOP)
    add_subdirectory(services)
    add_subdirectory(ui)
  endif()
  add_subdirectory(app)
endif()
if(BUILD_TESTING)
  include(tests/PortableProfileTests.cmake)
  biglybtc_profile_workflow_tests(biglybtc_profile "${BIGLYBTC_LOCAL_ENGINE}")
  if(BIGLYBTC_LOCAL_ENGINE)
    include(tests/PortableNetworkTests.cmake)
    biglybtc_add_portable_network_tests(biglybtc_local_engine biglybtc_mainline_bind_plan)
    biglybtc_add_portable_magnet_test(biglybtc_local_engine)
    biglybtc_add_portable_torrent_creator_test(biglybtc_local_engine)
  endif()
endif()
# The rest of the preview suite uses Linux fixtures and GNU linker fault injection.
# The native profile suite is configured separately from platform/.
if(BUILD_TESTING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_subdirectory(tests)
elseif(BUILD_TESTING AND BIGLYBTC_DESKTOP)
  find_package(Qt6 ${BIGLYBTC_QT_REQUIREMENT} REQUIRED COMPONENTS Test)
  add_executable(biglybtc-saved-rss-source-tests tests/saved_rss_source_tests.cpp)
  set_target_properties(biglybtc-saved-rss-source-tests PROPERTIES AUTOMOC ON)
  target_link_libraries(biglybtc-saved-rss-source-tests PRIVATE biglybtc_saved_rss_source Qt6::Test)
  add_test(NAME saved-rss-source-description COMMAND biglybtc-saved-rss-source-tests)
  if(WIN32)
    add_executable(biglybtc-scheduler-service-tests tests/scheduler_service_tests.cpp)
    target_link_libraries(biglybtc-scheduler-service-tests PRIVATE biglybtc_scheduler_service)
    if(MINGW)
      target_link_options(biglybtc-scheduler-service-tests PRIVATE -municode)
    endif()
    add_test(NAME scheduler-service COMMAND biglybtc-scheduler-service-tests "$<TARGET_FILE:biglybtc_scheduler_module>")
  endif()
endif()
include(packaging/Debian.cmake)
