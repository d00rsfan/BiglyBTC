cmake_minimum_required(VERSION 3.25)
project(BiglyBTC VERSION 0.1.0 LANGUAGES CXX)
include(cmake/Toolchain.cmake)
option(BIGLYBTC_DESKTOP "Build the experimental Qt desktop using native services" OFF)
enable_testing()
if(BIGLYBTC_DESKTOP)
  set(BIGLYBTC_SDK_TESTS OFF CACHE BOOL "SDK tested separately")
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../plugin-sdk" "plugin-sdk" EXCLUDE_FROM_ALL)
endif()
add_subdirectory(core)
add_subdirectory(adapters)
add_subdirectory(plugins/BiglyBTC-plugin-azrating)
option(BIGLYBTC_LOCAL_ENGINE "Build the experimental owned-loopback engine runner" OFF)
option(BIGLYBTC_DESKTOP "Build the experimental Qt desktop using native services" OFF)
if(BIGLYBTC_DESKTOP AND NOT BIGLYBTC_LOCAL_ENGINE)
  message(FATAL_ERROR "The experimental desktop currently requires the local engine adapter")
endif()
if(BIGLYBTC_LOCAL_ENGINE)
  add_subdirectory(engine)
  if(BIGLYBTC_DESKTOP)
    add_subdirectory(services)
    add_subdirectory(ui)
  endif()
  add_subdirectory(app)
endif()
add_subdirectory(tests)
include(packaging/Debian.cmake)
