cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(BiglyBTCPlatform LANGUAGES CXX)
  include(CTest)
endif()

add_library(biglybtc_profile_lock STATIC)
target_include_directories(biglybtc_profile_lock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(biglybtc_profile_lock PUBLIC cxx_std_23)
if(WIN32)
  target_sources(biglybtc_profile_lock PRIVATE profile_lock_windows.cpp)
  target_link_libraries(biglybtc_profile_lock PRIVATE advapi32)
else()
  target_sources(biglybtc_profile_lock PRIVATE profile_lock_posix.cpp)
endif()
if(MSVC)
  target_compile_options(biglybtc_profile_lock PRIVATE /W4 /WX)
else()
  target_compile_options(biglybtc_profile_lock PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(biglybtc_profile_lease STATIC ../adapters/profile_lease.cpp)
target_include_directories(biglybtc_profile_lease PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}/../adapters" "${CMAKE_CURRENT_SOURCE_DIR}/../core/include")
target_link_libraries(biglybtc_profile_lease PUBLIC biglybtc_profile_lock)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND BUILD_TESTING)
  find_package(Java 17 REQUIRED COMPONENTS Runtime)
  add_executable(biglybtc-profile-lock-probe profile_lock_probe.cpp)
  target_link_libraries(biglybtc-profile-lock-probe PRIVATE biglybtc_profile_lease)
  if(MINGW)
    target_link_options(biglybtc-profile-lock-probe PRIVATE -municode)
  endif()
  add_test(NAME java-profile-lock-interop
    COMMAND "${Java_JAVA_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/ProfileLockInterop.java"
      "$<TARGET_FILE:biglybtc-profile-lock-probe>")
  set_tests_properties(java-profile-lock-interop PROPERTIES TIMEOUT 45)

  if(UNIX)
    add_library(biglybtc-loaded-mapping-fixture SHARED loaded_mapping_fixture.cpp)
    add_executable(biglybtc-loaded-mapping-probe loaded_mapping_probe.cpp)
    target_compile_features(biglybtc-loaded-mapping-probe PRIVATE cxx_std_23)
    target_link_libraries(biglybtc-loaded-mapping-probe PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME executable-mapping-file-identity COMMAND biglybtc-loaded-mapping-probe
      "$<TARGET_FILE:biglybtc-loaded-mapping-fixture>")
    set_tests_properties(executable-mapping-file-identity PROPERTIES TIMEOUT 10)

    add_subdirectory(../core native-core EXCLUDE_FROM_ALL)
    include(../adapters/profile_storage.cmake)
    add_executable(biglybtc-profile-file-tests profile_file_tests.cpp)
    target_link_libraries(biglybtc-profile-file-tests PRIVATE biglybtc_profile)
    add_test(NAME profile-config-transactions COMMAND biglybtc-profile-file-tests)
    set_tests_properties(profile-config-transactions PROPERTIES TIMEOUT 15)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      add_executable(biglybtc-config-create-tests ../tests/config_create_tests.cpp)
      target_link_libraries(biglybtc-config-create-tests PRIVATE biglybtc_profile)
      target_link_options(biglybtc-config-create-tests PRIVATE "LINKER:--wrap=fsync" "LINKER:--wrap=renameat2")
      add_test(NAME compatible-config-first-create COMMAND biglybtc-config-create-tests)
      set_tests_properties(compatible-config-first-create PROPERTIES TIMEOUT 30)
    endif()
  endif()
endif()
