cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(BiglyBTCPlatform LANGUAGES CXX)
  include(CTest)
endif()

add_library(biglybtc_profile_lock STATIC)
target_include_directories(biglybtc_profile_lock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(biglybtc_profile_lock PUBLIC cxx_std_23)
if(WIN32)
  target_sources(biglybtc_profile_lock PRIVATE profile_lock_windows.cpp)
  target_link_libraries(biglybtc_profile_lock PRIVATE advapi32)
else()
  target_sources(biglybtc_profile_lock PRIVATE profile_lock_posix.cpp)
endif()
if(MSVC)
  target_compile_options(biglybtc_profile_lock PRIVATE /W4 /WX)
else()
  target_compile_options(biglybtc_profile_lock PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND BUILD_TESTING)
  find_package(Java 17 REQUIRED COMPONENTS Runtime)
  add_executable(biglybtc-profile-lock-probe profile_lock_probe.cpp)
  target_link_libraries(biglybtc-profile-lock-probe PRIVATE biglybtc_profile_lock)
  if(MINGW)
    target_link_options(biglybtc-profile-lock-probe PRIVATE -municode)
  endif()
  add_test(NAME java-profile-lock-interop
    COMMAND "${Java_JAVA_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/ProfileLockInterop.java"
      "$<TARGET_FILE:biglybtc-profile-lock-probe>")
  set_tests_properties(java-profile-lock-interop PROPERTIES TIMEOUT 45)
endif()
