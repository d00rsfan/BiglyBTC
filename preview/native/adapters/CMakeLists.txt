# Append to preview/native/adapters/CMakeLists.txt. Off by default: this is a
# read-only prerequisite, not activation of original wire identity.
option(BIGLYBTC_ORIGINAL_MLDHT_IDENTITY_READER "Build the source-bound original mlDHT root reader" OFF)
set(BIGLYBTC_MLDHT_IDENTITY_SOURCE_DIR "" CACHE PATH "Explicit reviewed BiglyBTC-plugin-mlDHT source directory")
if(BIGLYBTC_ORIGINAL_MLDHT_IDENTITY_READER)
  if(NOT IS_ABSOLUTE "${BIGLYBTC_MLDHT_IDENTITY_SOURCE_DIR}")
    message(FATAL_ERROR "The original mlDHT reader requires an explicit reviewed absolute plugin source path")
  endif()
  file(REAL_PATH "${BIGLYBTC_MLDHT_IDENTITY_SOURCE_DIR}" mldht_identity_source)
  file(SHA256 "${mldht_identity_source}/identity.cpp" mldht_identity_cpp_sha)
  file(SHA256 "${mldht_identity_source}/identity.hpp" mldht_identity_hpp_sha)
  if(NOT mldht_identity_cpp_sha STREQUAL "37a830751932c28daf8c99ddc17400b0f72cf65b5ad8ee9e406943e600cf74bc"
      OR NOT mldht_identity_hpp_sha STREQUAL "d1df373ebacb07dddc1736f3a93310c7318a22a33d31179c8b5d31de1e36f6de")
    message(FATAL_ERROR "Original mlDHT identity dependency differs from the qualified source")
  endif()
  if(TARGET biglybtc_mldht_identity)
    get_target_property(mldht_identity_existing_dir biglybtc_mldht_identity SOURCE_DIR)
    get_target_property(mldht_identity_existing_sources biglybtc_mldht_identity SOURCES)
    get_target_property(mldht_identity_existing_imported biglybtc_mldht_identity IMPORTED)
    get_target_property(mldht_identity_existing_type biglybtc_mldht_identity TYPE)
    list(LENGTH mldht_identity_existing_sources mldht_identity_source_count)
    if(mldht_identity_existing_imported OR NOT mldht_identity_existing_type STREQUAL "STATIC_LIBRARY"
        OR NOT mldht_identity_source_count EQUAL 1)
      message(FATAL_ERROR "Original mlDHT identity target must build the one reviewed source")
    endif()
    list(GET mldht_identity_existing_sources 0 mldht_identity_existing_cpp)
    file(REAL_PATH "${mldht_identity_existing_cpp}" mldht_identity_existing_cpp BASE_DIRECTORY "${mldht_identity_existing_dir}")
    if(NOT mldht_identity_existing_cpp STREQUAL "${mldht_identity_source}/identity.cpp")
      message(FATAL_ERROR "Original mlDHT identity target uses a different source path")
    endif()
  else()
    # Reuse the plugin source and its established target contract, no copied
    # parser/derivation algorithm or standalone plugin/runtime activation.
    add_library(biglybtc_mldht_identity STATIC "${mldht_identity_source}/identity.cpp")
    target_include_directories(biglybtc_mldht_identity PUBLIC "${mldht_identity_source}")
    target_compile_features(biglybtc_mldht_identity PUBLIC cxx_std_23)
    set_target_properties(biglybtc_mldht_identity PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(biglybtc_mldht_identity PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_library(biglybtc_mainline_identity STATIC mainline_identity.cpp)
  target_include_directories(biglybtc_mainline_identity PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(biglybtc_mainline_identity PUBLIC biglybtc_profile PRIVATE biglybtc_plugin_settings biglybtc_mldht_identity)
  target_compile_features(biglybtc_mainline_identity PUBLIC cxx_std_23)
  target_compile_options(biglybtc_mainline_identity PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
  message(STATUS "Original mlDHT root reader disabled explicitly; original wire identity activation is unavailable")
endif()
