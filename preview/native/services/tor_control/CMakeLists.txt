# Byte-exact service export. No Tor process/application activation.
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Network Test)
if(NOT TARGET biglybtc_tor_auth)
  message(FATAL_ERROR "Explicit qualified SAFECOOKIE target is required")
endif()
add_library(biglybtc_tor_channel STATIC channel.hpp channel.cpp)
set_target_properties(biglybtc_tor_channel PROPERTIES AUTOMOC ON)
target_include_directories(biglybtc_tor_channel PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(biglybtc_tor_channel PUBLIC Qt6::Core Qt6::Network PRIVATE biglybtc_tor_auth OpenSSL::Crypto)
target_compile_options(biglybtc_tor_channel PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(biglybtc-tor-channel-tests channel_tests.cpp)
set_target_properties(biglybtc-tor-channel-tests PROPERTIES AUTOMOC ON)
target_link_libraries(biglybtc-tor-channel-tests PRIVATE biglybtc_tor_channel Qt6::Test)
target_compile_options(biglybtc-tor-channel-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME owned-tor-control-channel COMMAND biglybtc-tor-channel-tests)
set_tests_properties(owned-tor-control-channel PROPERTIES TIMEOUT 120 LABELS "integration;loopback;qt")
add_library(biglybtc_tor_ownership STATIC ownership.hpp ownership.cpp)
set_target_properties(biglybtc_tor_ownership PROPERTIES AUTOMOC ON)
