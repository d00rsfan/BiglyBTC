cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(BiglyBTC-plugin-azrating VERSION 0.1.0 LANGUAGES CXX)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Toolchain.cmake")
    include(cmake/Toolchain.cmake)
  else()
    include(../../cmake/Toolchain.cmake)
  endif()
  enable_testing()
endif()
if(NOT TARGET BiglyBTC::Core)
  find_package(BiglyBTCCore 0.1.0 EXACT CONFIG REQUIRED)
endif()
add_library(biglybtc_rating_data STATIC rating_data.cpp local_rating.cpp)
target_include_directories(biglybtc_rating_data PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(biglybtc_rating_data PUBLIC BiglyBTC::Core)
target_compile_features(biglybtc_rating_data PUBLIC cxx_std_23)
set_target_properties(biglybtc_rating_data PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(biglybtc_rating_data PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
add_executable(biglybtc-rating-driver rating_driver.cpp)
target_link_libraries(biglybtc-rating-driver PRIVATE biglybtc_rating_data)
target_compile_options(biglybtc-rating-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME original-rating-data COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/fixture_test.py" "$<TARGET_FILE:biglybtc-rating-driver>")
set_tests_properties(original-rating-data PROPERTIES TIMEOUT 90 LABELS "unit;fixtures")
add_executable(biglybtc-rating-bounds-tests bounds_test.cpp)
target_link_libraries(biglybtc-rating-bounds-tests PRIVATE biglybtc_rating_data)
target_compile_options(biglybtc-rating-bounds-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded-rating-data COMMAND biglybtc-rating-bounds-tests)
set_tests_properties(bounded-rating-data PROPERTIES TIMEOUT 30 LABELS "unit")
add_executable(biglybtc-rating-local-driver local_driver.cpp)
target_link_libraries(biglybtc-rating-local-driver PRIVATE biglybtc_rating_data)
target_compile_options(biglybtc-rating-local-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME original-rating-local-policy COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/local_fixture_test.py" "$<TARGET_FILE:biglybtc-rating-local-driver>")
set_tests_properties(original-rating-local-policy PROPERTIES TIMEOUT 120 LABELS "unit;fixtures")
add_executable(biglybtc-rating-local-bounds-tests local_bounds_test.cpp)
target_link_libraries(biglybtc-rating-local-bounds-tests PRIVATE biglybtc_rating_data)
target_compile_options(biglybtc-rating-local-bounds-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded-rating-local-policy COMMAND biglybtc-rating-local-bounds-tests)
set_tests_properties(bounded-rating-local-policy PROPERTIES TIMEOUT 30 LABELS "unit")
