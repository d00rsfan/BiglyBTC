#!/usr/bin/env python3
import os, sys
from pathlib import Path
root = Path(__file__).resolve().parent
private = root / 'usr/lib/biglybtc'
lookup = root / 'dependencies/libtorrent/usr/lib/x86_64-linux-gnu'
os.environ['LD_LIBRARY_PATH'] = os.pathsep.join(map(str, [private / 'lib', private / 'qt/lib', lookup]))
os.environ['QT_PLUGIN_PATH'] = str(private / 'qt/plugins')
os.environ['QT_QPA_PLATFORM_PLUGIN_PATH'] = str(private / 'qt/plugins/platforms')
os.execv(str(private / 'bin/BiglyBTC'), ['BiglyBTC', *sys.argv[1:]])
