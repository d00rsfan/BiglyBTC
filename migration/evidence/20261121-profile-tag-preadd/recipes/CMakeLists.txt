cmake_minimum_required(VERSION 3.25)
project(BiglyBTCProfileSharedTagRates LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Test)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(profile-cohort STATIC drafts/profile_tag_rates.cpp)
target_include_directories(profile-cohort PUBLIC drafts "${NATIVE_ROOT}/engine" "${NATIVE_ROOT}/adapters" "${NATIVE_ROOT}/core/include")
target_compile_definitions(profile-cohort PUBLIC BIGLYBTC_SCALABLE_TAG_RATE_RUNTIME=1)
target_compile_options(profile-cohort PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(profile-cohort PUBLIC LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS} "${PREADD_LIBRARY}"
 "$<LINK_GROUP:RESCAN,${NATIVE_CACHE}/adapters/libbiglybtc_profile.a,${NATIVE_CACHE}/adapters/libbiglybtc_state_codec.a,${NATIVE_CACHE}/core/libbiglybtc_core.a,${NATIVE_CACHE}/adapters/libbiglybtc_subscription_automatic_values.a,${NATIVE_CACHE}/engine/libbiglybtc_subscription_download_values.a>" ZLIB::ZLIB OpenSSL::Crypto Qt6::Core)
add_executable(profile-cohort-tests "${NATIVE_ROOT}/tests/profile_cohort_tests.cpp")
target_link_libraries(profile-cohort-tests PRIVATE profile-cohort Qt6::Test)
target_compile_options(profile-cohort-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(profile-existing-source-tests "${NATIVE_ROOT}/tests/prepared_tag_rate_source_tests.cpp")
target_link_libraries(profile-existing-source-tests PRIVATE profile-cohort Qt6::Test)
target_compile_options(profile-existing-source-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(profile-cohort-payload profile_payload_tests.cpp)
target_include_directories(profile-cohort-payload PRIVATE "${TRANSPORT_FIXTURE}")
target_link_libraries(profile-cohort-payload PRIVATE profile-cohort)
target_compile_options(profile-cohort-payload PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(profile-pending-controls profile_preadd_tests.cpp)
target_include_directories(profile-pending-controls PRIVATE "${NATIVE_ROOT}/tests")
target_link_libraries(profile-pending-controls PRIVATE profile-cohort)
target_compile_options(profile-pending-controls PRIVATE -Wall -Wextra -Wpedantic -Werror)
