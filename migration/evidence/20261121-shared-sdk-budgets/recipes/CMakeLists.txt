cmake_minimum_required(VERSION 3.25)
project(BiglyBTCSharedTagBudget LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(shared-session session_tests.cpp)
target_link_libraries(shared-session PRIVATE LibtorrentRasterbar::torrent-rasterbar Threads::Threads)
target_compile_options(shared-session PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(shared-native STATIC native-drafts/sdk_shared_tag_budget.cpp native-drafts/sdk_tag_rate_group.cpp)
target_include_directories(shared-native PUBLIC native-drafts native-drafts/include "${NATIVE_ROOT}/core/include")
target_link_libraries(shared-native PUBLIC LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(shared-native PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(shared-wrapper wrapper_tests.cpp)
target_link_libraries(shared-wrapper PRIVATE shared-native)
target_compile_options(shared-wrapper PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(shared-wrapper-old-headers wrapper_tests.cpp native-drafts/sdk_shared_tag_budget.cpp native-drafts/sdk_tag_rate_group.cpp)
target_include_directories(shared-wrapper-old-headers PRIVATE native-drafts native-drafts/include "${NATIVE_ROOT}/core/include")
target_link_libraries(shared-wrapper-old-headers PRIVATE LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(shared-wrapper-old-headers PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(shared-wrapper-old-headers SYSTEM BEFORE PRIVATE "${OLD_SDK_INCLUDE}")
add_library(shared-image-driver SHARED image_driver.cpp)
target_link_libraries(shared-image-driver PRIVATE LibtorrentRasterbar::torrent-rasterbar Threads::Threads)
target_compile_options(shared-image-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(shared-images image_tests.cpp)
target_link_libraries(shared-images PRIVATE LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(shared-images PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(shared-interposer SHARED shared_interposer.cpp)
add_executable(shared-backend-transport backend_transport_tests.cpp)
target_link_libraries(shared-backend-transport PRIVATE shared-native)
target_compile_options(shared-backend-transport PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(ordinary-readiness "${NATIVE_ROOT}/tests/sdk_tag_rate_readiness_tests.cpp")
target_link_libraries(ordinary-readiness PRIVATE shared-native)
target_compile_options(ordinary-readiness PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(ordinary-payload "${NATIVE_ROOT}/tests/sdk_tag_rate_group_tests.cpp")
target_link_libraries(ordinary-payload PRIVATE shared-native)
target_compile_options(ordinary-payload PRIVATE -Wall -Wextra -Wpedantic -Werror)
set(SHARED_SDK_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/sdk-source" CACHE PATH "Exact SDK implementation under qualification")
add_library(foreign-allocator SHARED foreign_allocator_driver.cpp
 "${SHARED_SDK_SOURCE}/src/biglybtc_shared_quota.cpp"
 "${SHARED_SDK_SOURCE}/src/biglybtc_shared_binding.cpp"
 "${SHARED_SDK_SOURCE}/src/biglybtc_tag_shared.cpp"
 "${SHARED_SDK_SOURCE}/src/biglybtc_tag_gate.cpp")
target_link_libraries(foreign-allocator PRIVATE LibtorrentRasterbar::torrent-rasterbar Threads::Threads)
target_compile_definitions(foreign-allocator PRIVATE TORRENT_BUILDING_LIBRARY TORRENT_BUILDING_SHARED)
target_link_options(foreign-allocator PRIVATE -Wl,-Bsymbolic-functions)
set_target_properties(foreign-allocator PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
# Test driver functions are explicit C exports;the allocator/binder exports
# already carry their production header's visibility declarations.
set_source_files_properties(foreign_allocator_driver.cpp PROPERTIES COMPILE_OPTIONS "-fvisibility=default")
add_executable(allocator-images allocator_image_tests.cpp)
target_link_libraries(allocator-images PRIVATE LibtorrentRasterbar::torrent-rasterbar ${CMAKE_DL_LIBS})
target_compile_options(allocator-images PRIVATE -Wall -Wextra -Wpedantic -Werror)
