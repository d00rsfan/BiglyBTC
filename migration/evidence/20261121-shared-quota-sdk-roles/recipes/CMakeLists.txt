cmake_minimum_required(VERSION 3.25)
project(BiglyBTCSharedRoleQuota LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(role-quota role_quota_controls.cpp)
target_include_directories(role-quota PRIVATE "${NATIVE_ROOT}/engine" "${NATIVE_ROOT}/core/include" "${TRANSPORT_FIXTURE}")
target_link_libraries(role-quota PRIVATE "${SHARED_WRAPPER}" LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(role-quota PRIVATE -Wall -Wextra -Wpedantic -Werror)
