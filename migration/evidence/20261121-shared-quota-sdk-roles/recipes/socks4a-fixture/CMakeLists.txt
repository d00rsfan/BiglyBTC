cmake_minimum_required(VERSION 3.24)
project(BiglyBTCSocks4aProviderQualification LANGUAGES CXX)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Network Test)
add_executable(biglybtc-socks4a-provider-tests socks4a_provider_tests.cpp)
set_target_properties(biglybtc-socks4a-provider-tests PROPERTIES AUTOMOC ON)
target_compile_features(biglybtc-socks4a-provider-tests PRIVATE cxx_std_23)
target_compile_options(biglybtc-socks4a-provider-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(biglybtc-socks4a-provider-tests PRIVATE LibtorrentRasterbar::torrent-rasterbar Qt6::Core Qt6::Network Qt6::Test ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME socks4a-provider COMMAND biglybtc-socks4a-provider-tests)
set_tests_properties(socks4a-provider PROPERTIES TIMEOUT 90)
