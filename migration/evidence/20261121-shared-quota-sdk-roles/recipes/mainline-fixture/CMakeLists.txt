cmake_minimum_required(VERSION 3.28)
project(OwnedMainlinePortProof LANGUAGES CXX)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IS_ABSOLUTE "${MAINLINE_PROVIDER_INSTALL}" OR NOT IS_ABSOLUTE "${MAINLINE_PROVIDER_SOURCE}")
  message(FATAL_ERROR "Explicit absolute MAINLINE_PROVIDER_INSTALL/SOURCE required")
endif()
file(SHA256 "${MAINLINE_PROVIDER_SOURCE}/include/libtorrent/biglybtc_mainline_port.hpp" source_hash)
file(SHA256 "${MAINLINE_PROVIDER_INSTALL}/include/libtorrent/biglybtc_mainline_port.hpp" installed_hash)
if(NOT source_hash STREQUAL installed_hash)
  message(FATAL_ERROR "Mainline bridge source/installed SDK mismatch")
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY FALSE)
find_package(LibtorrentRasterbar CONFIG REQUIRED PATHS "${MAINLINE_PROVIDER_INSTALL}/lib/cmake/LibtorrentRasterbar" NO_DEFAULT_PATH)
find_package(Threads REQUIRED)
add_executable(owned-mainline-port-proof mainline_port_proof.cpp)
target_compile_options(owned-mainline-port-proof PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(owned-mainline-port-proof PRIVATE MAINLINE_EXPECTED_PROVIDER="${MAINLINE_PROVIDER_INSTALL}/lib/libtorrent-rasterbar.so.2.0.12")
target_link_libraries(owned-mainline-port-proof PRIVATE LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS})
enable_testing()
add_test(NAME mainline-port-bridge COMMAND owned-mainline-port-proof "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(mainline-port-bridge PROPERTIES TIMEOUT 90)
