cmake_minimum_required(VERSION 3.25)
project(BiglyBTCSharedRoleRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(role-runtime ../runtime_probe.cpp "${NATIVE_ROOT}/engine/persistent_runtime_guard.cpp" "${NATIVE_ROOT}/services/i2p_provider/owner_runtime_guard.cpp")
target_include_directories(role-runtime PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/binding" "${NATIVE_ROOT}/engine" "${NATIVE_ROOT}/services/i2p_provider")
target_link_libraries(role-runtime PRIVATE LibtorrentRasterbar::torrent-rasterbar OpenSSL::Crypto ${CMAKE_DL_LIBS})
target_compile_options(role-runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
