cmake_minimum_required(VERSION 3.25)
project(BiglyBTCPublicSharedRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(shared-runtime provider_probe.cpp drafts/persistent_runtime_guard.cpp)
target_include_directories(shared-runtime PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/binding" drafts "${NATIVE_ROOT}/engine")
target_link_libraries(shared-runtime PRIVATE LibtorrentRasterbar::torrent-rasterbar OpenSSL::Crypto ${CMAKE_DL_LIBS})
target_compile_options(shared-runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(shared-mappings "${NATIVE_ROOT}/tests/loaded_mapping_snapshot_tests.cpp")
target_include_directories(shared-mappings PRIVATE drafts)
target_compile_options(shared-mappings PRIVATE -Wall -Wextra -Wpedantic -Werror)
