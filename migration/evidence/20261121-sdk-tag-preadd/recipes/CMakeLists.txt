cmake_minimum_required(VERSION 3.25)
project(BiglyBTCTagPreAdd LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(preadd STATIC drafts/sdk_tag_rate_group.cpp "${NATIVE_ROOT}/engine/sdk_shared_tag_budget.cpp")
target_include_directories(preadd PUBLIC drafts "${NATIVE_ROOT}/engine" "${NATIVE_ROOT}/core/include")
target_link_libraries(preadd PUBLIC LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(preadd PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(preadd-controls "${NATIVE_ROOT}/tests/sdk_preadd_tests.cpp")
target_include_directories(preadd-controls PRIVATE "${TRANSPORT_FIXTURE}")
target_link_libraries(preadd-controls PRIVATE preadd)
target_compile_options(preadd-controls PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(preadd-payload preadd_payload_tests.cpp)
target_include_directories(preadd-payload PRIVATE "${TRANSPORT_FIXTURE}")
target_link_libraries(preadd-payload PRIVATE preadd)
target_compile_options(preadd-payload PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(preadd-existing "${TRANSPORT_FIXTURE}/wrapper_tests.cpp")
target_link_libraries(preadd-existing PRIVATE preadd)
target_compile_options(preadd-existing PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(preadd-workers "${CMAKE_CURRENT_SOURCE_DIR}/../tag-rate-worker-bridge-20261121/worker_control_tests.cpp")
target_link_libraries(preadd-workers PRIVATE preadd)
target_compile_options(preadd-workers PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(preadd-edges "${NATIVE_ROOT}/tests/sdk_preadd_edge_tests.cpp")
target_include_directories(preadd-edges PRIVATE "${TRANSPORT_FIXTURE}")
target_link_libraries(preadd-edges PRIVATE preadd)
target_compile_options(preadd-edges PRIVATE -Wall -Wextra -Wpedantic -Werror)
