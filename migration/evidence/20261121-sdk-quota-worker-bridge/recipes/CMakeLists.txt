cmake_minimum_required(VERSION 3.25)
project(BiglyBTCSharedQuotaWorkers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(worker-shared STATIC drafts/sdk_shared_tag_budget.cpp drafts/sdk_tag_rate_group.cpp)
target_include_directories(worker-shared PUBLIC drafts "${NATIVE_ROOT}/core/include")
target_link_libraries(worker-shared PUBLIC LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(worker-shared PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(worker-controls worker_control_tests.cpp)
target_link_libraries(worker-controls PRIVATE worker-shared)
target_compile_options(worker-controls PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(worker-baseline worker_control_tests.cpp)
target_include_directories(worker-baseline PRIVATE "${NATIVE_ROOT}/engine" "${NATIVE_ROOT}/core/include")
target_link_libraries(worker-baseline PRIVATE "${BASELINE_WRAPPER}" LibtorrentRasterbar::torrent-rasterbar Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(worker-baseline PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(worker-payload worker_payload_tests.cpp)
target_include_directories(worker-payload PRIVATE "${TRANSPORT_FIXTURE}")
target_link_libraries(worker-payload PRIVATE worker-shared)
target_compile_options(worker-payload PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(worker-existing "${WRAPPER_REFERENCE}")
target_link_libraries(worker-existing PRIVATE worker-shared)
target_compile_options(worker-existing PRIVATE -Wall -Wextra -Wpedantic -Werror)
